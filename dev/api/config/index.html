    Configuration

    All four dataclasses are frozen (immutable) and validate their inputs on construction. Import them from the top-level package or from aegiss.config_dataclasses:

    from aegiss import DMRGConfig, EntropyConfig, ProjectionConfig, OutputConfig
# or
from aegiss.config_dataclasses import DMRGConfig, EntropyConfig, ProjectionConfig, OutputConfig

    @dataclass(frozen=True)
class DMRGConfig:
    """Configuration dataclass for DMRG calculations.

    Args:
        ncas (int):
            Number of active orbitals for the DMRG calculation.
        nelecas (int or tuple[int, int]):
            Number of active electrons, either as a total count or an explicit
            ``(nalpha, nbeta)`` pair (same convention as PySCF's CASCI).
        max_bond_dim (int, default 500):
            Target DMRG bond dimension (must be a positive multiple of 4).
            The sweep schedule starts at ``max_bond_dim // 2`` and ramps up to ``max_bond_dim``.
        symm_type ({'SZ', 'SU2'}, default 'SZ'):
            Spin symmetry used by block2. Both give spin-summed single-orbital
            entropies directly from the ``1pdm``/``1npc`` output; ``'SU2'`` is
            spin-adapted and faster. ``'SZ'`` is required only for
            ``spin_soe='resolved'`` and for unrestricted (UHF) integrals.
        verbosity (int, default 0):
            Verbosity level for block2 output.  Higher values produce more detailed logs.
        unrestricted (bool or None, default None):
            Whether to emit an unrestricted (UHF) FCIDUMP. ``None`` autodetects from
            the mean-field type (``isinstance(mf, scf.uhf.UHF)``); pass ``True``/
            ``False`` to override. UHF requires ``symm_type='SZ'``.
        reorder ({'noreorder', 'fiedler'}, default 'noreorder'):
            Orbital ordering for block2main. ``'noreorder'`` keeps the FCIDUMP order
            and enables ``full_fci_space``; ``'fiedler'`` reorders by the Fiedler
            vector and disables ``full_fci_space``.
        spin_soe ({'auto', 'summed', 'resolved'}, default 'auto'):
            Whether single-orbital entropies (and the selection built on them) are
            spin-summed or spin-resolved. ``'summed'`` uses the 4-state spatial
            entropy, so both spin channels share one entropy window; for a UHF
            reference the AO projection still returns a split ``{'alpha', 'beta'}``
            active space. ``'resolved'`` uses the
            per-spin-orbital entropies ``H(n_alpha)`` / ``H(n_beta)`` and screens the
            alpha and beta channels independently, yielding a split
            ``{'alpha', 'beta'}`` active space. ``'auto'`` picks ``'resolved'`` when
            the target is spin-polarised (``nalpha != nbeta``) and ``symm_type='SZ'``,
            else ``'summed'``. ``'resolved'`` requires ``symm_type='SZ'`` (the
            spin-resolved 1-PDM is only available without spin adaptation).
    """

    ncas: int
    nelecas: int | tuple[int, int]
    max_bond_dim: int = 500
    symm_type: Literal["SZ", "SU2"] = "SZ"
    verbosity: int = 0
    unrestricted: bool | None = None
    reorder: Literal["noreorder", "fiedler"] = "noreorder"
    spin_soe: Literal["auto", "summed", "resolved"] = "auto"

    def __post_init__(self) -> None:
        """Validate the configuration parameters.

        Raises:
            ValueError: If any of the parameters are out of their expected ranges or have invalid values.
        """
        self._validate_cas()
        self._validate_solver()

    def _validate_cas(self) -> None:
        """Validate the active-space size and electron count."""
        if self.ncas <= 0:
            raise ValueError(
                f"DMRGConfig: Number of active orbitals (`ncas`) must be positive, got {self.ncas}."
            )
        if isinstance(self.nelecas, tuple) and len(self.nelecas) != 2:
            raise ValueError(
                "DMRGConfig: `nelecas` tuple must have exactly two elements (nalpha, nbeta)."
            )
        if isinstance(self.nelecas, int) and self.nelecas <= 0:
            raise ValueError(
                f"DMRGConfig: Number of active electrons (`nelecas`) must be positive, got {self.nelecas}."
            )
        nalpha, nbeta = resolve_nelecas(self.nelecas)
        if nalpha < nbeta or nbeta < 0:
            raise ValueError(
                f"DMRGConfig: `nelecas` must satisfy nalpha >= nbeta >= 0, got {self.nelecas}."
            )

    def _validate_solver(self) -> None:
        """Validate the bond dimension, symmetry, verbosity, and ordering options."""
        if self.max_bond_dim <= 0:
            raise ValueError(
                f"DMRGConfig: Target bond dimension (`max_bond_dim`) must be positive, got {self.max_bond_dim}."
            )
        if self.max_bond_dim % 4 != 0:
            raise ValueError(
                f"DMRGConfig: `max_bond_dim` must be a multiple of 4, got {self.max_bond_dim}."
            )
        if self.symm_type not in ("SZ", "SU2"):
            raise ValueError(
                f"DMRGConfig: Invalid symmetry type (`symm_type`) {self.symm_type!r}. Must be 'SZ' or 'SU2'."
            )
        if self.verbosity < 0:
            raise ValueError(
                f"DMRGConfig: `verbosity` must be non-negative, got {self.verbosity}."
            )
        if self.reorder not in ("noreorder", "fiedler"):
            raise ValueError(
                f"DMRGConfig: Invalid reorder {self.reorder!r}. "
                "Must be 'noreorder' or 'fiedler'."
            )
        if self.unrestricted is True and self.symm_type != "SZ":
            raise ValueError(
                "DMRGConfig: `unrestricted=True` requires `symm_type='SZ'`, "
                f"got {self.symm_type!r}."
            )
        if self.spin_soe not in ("auto", "summed", "resolved"):
            raise ValueError(
                f"DMRGConfig: Invalid spin_soe {self.spin_soe!r}. "
                "Must be 'auto', 'summed', or 'resolved'."
            )
        if self.spin_soe == "resolved" and self.symm_type != "SZ":
            raise ValueError(
                "DMRGConfig: `spin_soe='resolved'` requires `symm_type='SZ'`, "
                f"got {self.symm_type!r} (the spin-resolved 1-PDM is only "
                "available without spin adaptation)."
            )

    def spin_resolved(self) -> bool:
        """Resolve *spin_soe* to a concrete spin-resolved / spin-summed choice.

        ``'auto'`` selects spin-resolved only for a spin-polarised target
        (``nalpha != nbeta``) computed with ``symm_type='SZ'``; otherwise the
        spin-summed spatial entropy is used.

        Returns:
            spin_resolved (bool): ``True`` if the single-orbital entropies
                should be spin-resolved.
        """
        if self.spin_soe == "resolved":
            return True
        if self.spin_soe == "summed":
            return False
        nalpha, nbeta = resolve_nelecas(self.nelecas)
        return nalpha != nbeta and self.symm_type == "SZ"

class EntropyConfig:
    """Configuration dataclass for entropy-based active space screening.

    Args:
        fraction (float, default 0.1):
            Entropy threshold value (see *fraction_type*).
        fraction_type ({'r', 'a'}, default 'r'):
            ``'r'``: keep orbitals whose S(1) >= ``fraction x max(S(1))``;
            ``'a'``: keep orbitals whose S(1) ≥ ``fraction`` (absolute).
        ranking ({'subset', 'global'}, default 'subset'):
            How to rank orbitals for entropy screening.
            ``'subset'`` applies the threshold independently to occupied and virtual subspaces;
            ``'global'`` ranks all active orbitals together.
    """

    fraction: float = 0.1
    fraction_type: Literal["r", "a"] = "r"
    ranking: Literal["subset", "global"] = "subset"

    def __post_init__(self) -> None:
        """Validate the configuration parameters.

        Raises:
            ValueError: If any of the parameters are out of their expected ranges or have invalid values.
        """
        if not isinstance(self.fraction, (int, float)):
            raise ValueError(  # noqa: TRY004 -- documented as ValueError
                f"EntropyConfig: `fraction` must be a number, got {type(self.fraction).__name__}."
            )

        if self.fraction < 0 and self.fraction_type == "a":
            raise ValueError(
                f"EntropyConfig: `fraction` must be non-negative when `fraction_type` is 'a', got {self.fraction}."
            )
        if self.fraction_type not in ("r", "a"):
            raise ValueError(
                f"EntropyConfig: Invalid `fraction_type` {self.fraction_type!r}. Must be 'r' or 'a'."
            )
        if not (0 <= self.fraction <= 1) and self.fraction_type == "r":
            raise ValueError(
                f"EntropyConfig: `fraction` must be in the range [0, 1], got {self.fraction}."
            )
        if self.ranking not in ("subset", "global"):
            raise ValueError(
                f"EntropyConfig: Invalid `ranking` {self.ranking!r}. Must be 'subset' or 'global'."
            )

class ProjectionConfig:
    """Configuration dataclass for orbital selection based on atomic orbital projection.

    Args:
        ao_labels (list[str | list[str]]):
            AO labels to project onto. Each entry is either a single label string
            (e.g. ``"Fe 3d"``) or a list of strings treated as one group
            (e.g. ``["Fe 3d", "Fe 4d"]``).
        basis (str or None, default 'minao'):
            Basis set used for AO projection.  ``None`` uses the original
            molecular basis.
        contribution (float or list[float], default 0.5):
            Projection weight threshold (see *contribution_type*). A list specifies
            per-label thresholds and must be non-empty; if its length does not match
            *ao_labels*, the first value is broadcast to all labels.
        contribution_type ({'a', 'r'}, default 'a'):
            ``'a'``: absolute threshold;
            ``'r'``: relative to the maximum weight.
        top_n (int | list[int] |  None, default None):
            If set, keep the top-N orbitals per AO label instead of using a
            contribution threshold. If a list is provided, it must be non-empty and either
            match the length of *ao_labels* or be broadcastable to it.
        ranking ({'subset', 'global'}, default 'subset'):
            ``'subset'``: run AO projection independently on the occupied and virtual entropy-screened subspaces;
            ``'global'``: run it on the combined set.
    """

    ao_labels: list[str | list[str]]
    basis: str | None = "minao"
    contribution: float | list[float] = 0.5
    contribution_type: Literal["a", "r"] = "a"
    top_n: int | list[int] | None = None
    ranking: Literal["subset", "global"] = "subset"

    def __post_init__(self) -> None:
        """Validate the configuration parameters.

        Raises:
            ValueError: If any of the parameters are out of their expected ranges or have invalid values.
        """
        if not isinstance(self.ao_labels, list):
            raise ValueError(  # noqa: TRY004 -- documented as ValueError
                "ProjectionConfig: `ao_labels` must be a non-empty list."
            )
        if len(self.ao_labels) == 0:
            raise ValueError("ProjectionConfig: `ao_labels` must be a non-empty list.")

        if not (
            isinstance(self.ao_labels, list)
            and self.ao_labels
            and all(_valid_ao_label(label) for label in self.ao_labels)
        ):
            raise ValueError(
                "ProjectionConfig: `ao_labels` must be a non-empty list where each "
                "entry is a non-empty string or a non-empty list of strings."
            )
        if self.contribution_type not in ("a", "r"):
            raise ValueError(
                "ProjectionConfig: Invalid `contribution_type` "
                f"{self.contribution_type!r}. Must be 'a' or 'r'."
            )
        if self.ranking not in ("subset", "global"):
            raise ValueError(
                f"ProjectionConfig: Invalid `ranking` {self.ranking!r}. Must be 'subset' or 'global'."
            )

        _helper_contributions(self.contribution, self.contribution_type)

        if self.top_n is not None:
            _helper_top_n(top_n=self.top_n)

    """Configuration dataclass for AEGISS output options.

    Args:
        saving_folder (str, default './aegiss_output/'):
            Directory where cube files are written when *save_orb* is ``True``.
        saving_prefix (str, default 'aegiss_selection'):
            Prefix used for output cube file names.
        save_plots (bool, default False):
            Whether to generate and save diagnostic plots: the SOE heatmap and
            TH plot (spin-summed entropies only), the AO-contribution heatmaps of
            the projection step, and the AO pie chart of the final active
            orbitals (restricted references only).
        save_orb (bool, default True):
            Write cube files for the entropy-screened and final active orbitals.
    """

    saving_folder: str = "./aegiss_output/"
    saving_prefix: str = "aegiss_selection"
    save_orb: bool = True
    save_plots: bool = False

    def __post_init__(self) -> None:
        """Validate the configuration parameters.

        Raises:
            ValueError: If any of the parameters are out of their expected ranges or have invalid values.
        """
        if not self.saving_folder:
            raise ValueError(
                "OutputConfig: `saving_folder` must be a non-empty string."
            )
        if not self.saving_prefix:
            raise ValueError(
                "OutputConfig: `saving_prefix` must be a non-empty string."
            )
