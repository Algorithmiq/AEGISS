    Localization

    Orbital localization methods. localize is a dispatcher that accepts a method name as its first argument; each specific function can also be called directly.

    from aegiss.localization import localize

    Localized orbitals and the DMRG window

    The DMRG active region is a window of ncas orbitals taken by index around the occupied/virtual boundary. For canonical orbitals that is a Fermi-level window, because their index order follows the orbital energies. Localization rotates the occupied and virtual blocks separately: the occupied/virtual partition survives, but localized orbitals have no orbital energies and their order within each block is arbitrary, so the same index window no longer selects the frontier orbitals. Natural orbitals are ordered by occupation number, a different but meaningful criterion. Localized orbitals can still be used, but check that the window contains the orbitals you care about. See Methodology.

    def localize(method: str, *args: Any, **kwargs: Any) -> Any:
    """Dispatch to a localization method by name.

    Args:
        method (str):
            One of ``'boys'``, ``'edmiston'`` / ``'er'``, ``'pm'`` / ``'pipek'``,
            ``'ibos'`` / ``'ibo'``, ``'iaos'`` / ``'iao'``, ``'nos'``, ``'fnos'``.
        *args:
            Positional arguments forwarded to the selected method.
        **kwargs:
            Keyword arguments forwarded to the selected method.

    Raises:
        ValueError: If *method* is not one of the names listed above.

    Note:
        Localized orbitals have no orbital energies, so the index-based DMRG window
        (``ncas`` orbitals around the occupied/virtual boundary) is no longer a
        Fermi-level window when built on them.

    Example:
        >>> localize("pm", mol, mf, on_virtual=True)
        >>> localize("ibos", mol, mf)
        >>> localize("iaos", mol, mf, minao="minao")
        >>> localize("nos", mf, correlated_method="mp2")
        >>> localize("fnos", mf, correlated_method="mp2", frozen=[0])
        >>> localize("boys", mol, mf, n_occ=5, on_virtual=False)
    """
    if method not in _METHODS:
        raise ValueError(
            f"Unknown localization method {method!r}. Choose from: {sorted(_METHODS)}"
        )
    return _METHODS[method](*args, **kwargs)

    def localize_boys(
    mol: gto.Mole,
    mf: Any,
    n_occ: int | None = None,
    *,
    on_virtual: bool = False,
) -> np.ndarray:
    """Boys (Foster-Boys) localization — minimises the spatial spread of each orbital.

    Args:
        mol (pyscf.gto.Mole):
            Molecular geometry and basis set.
        mf (pyscf.scf.hf.SCF):
            Converged mean-field object supplying ``mo_coeff`` and ``mo_occ``.
        n_occ (int, optional):
            Number of occupied orbitals. Inferred from *mf* if not given.
        on_virtual (bool):
            Whether to also localize the virtual space.

    Returns:
        new_mos (np.ndarray, shape (nao, nmo)):
            Localized MO coefficients.
    """
    if n_occ is None:
        n_occ = int(np.count_nonzero(mf.mo_occ > 0))
    loc_occ = lo.Boys(mol, mf.mo_coeff[:, :n_occ]).kernel()
    loc_virt = (
        lo.Boys(mol, mf.mo_coeff[:, n_occ:]).kernel()
        if on_virtual
        else mf.mo_coeff[:, n_occ:]
    )
    return np.hstack((loc_occ, loc_virt))

    :func:`localize_fnos` instead.

    Args:
        mf (pyscf.scf.hf.SCF):
            Converged mean-field object.

            separately instead of the full matrix.

        nos (np.ndarray, shape (nao, nmo)):

    from aegiss.localization.nos import compute_nos  # noqa: PLC0415

    logger.info("%s NOs over the full orbital space.", "Split" if split else "Standard")
    result: np.ndarray = compute_nos(wfn_obj, split_nos=split)
    return result

    def localize_ibos(
    mol: gto.Mole,
    mf: Any,
    n_occ: int | None = None,
    *,
    minao: str = "minao",
    exponent: int = 4,
) -> np.ndarray:
    """Intrinsic Bond Orbital (IBO) localization.

    Localizes occupied MOs by maximising their population on intrinsic atomic
    orbitals (IAOs). IBOs tend to give chemically intuitive bond and lone-pair
    orbitals, and handle systems with mixed ionic/covalent character well.

    Args:
        mol (pyscf.gto.Mole):
            Molecular geometry and basis set.
        mf (pyscf.scf.hf.SCF):
            Converged mean-field object supplying ``mo_coeff`` and ``mo_occ``.
        n_occ (int, optional):
            Number of occupied orbitals. Inferred from *mf* if not given.
        minao (str):
            Reference minimal basis used for IAO construction (default ``"minao"``).
        exponent (int):
            Exponent in the IBO localization criterion (default ``4``).

    Returns:
        new_mos (np.ndarray, shape (nao, nmo)):
            IBO coefficients for the occupied block; virtual block is unchanged.
    """
    from pyscf.lo import ibo as _ibo  # noqa: PLC0415

    if mol.natm == 1:
        logger.warning(
            "IBO localization skipped: single-atom molecule has no bonds to localize."
        )
        return np.array(mf.mo_coeff)
    if n_occ is None:
        n_occ = int(np.count_nonzero(mf.mo_occ > 0))
    loc_occ = _ibo.ibo(mol, mf.mo_coeff[:, :n_occ], minao=minao, exponent=exponent)
    return np.hstack((loc_occ, mf.mo_coeff[:, n_occ:]))

    def localize_iaos(
    mol: gto.Mole,
    mf: Any,
    n_occ: int | None = None,
    *,
    minao: str = "minao",
) -> np.ndarray:
    """Intrinsic Atomic Orbital (IAO) construction.

    Constructs orthonormal IAOs that closely resemble minimal-basis atomic
    orbitals while spanning the occupied MO space. IAOs are useful as a
    chemically meaningful occupied basis for population analysis and as a
    starting point for IBO construction.

    There is one IAO per function of the reference minimal basis, which is in
    general more than the number of occupied orbitals, so the result is a basis
    for the occupied space and cannot replace ``mf.mo_coeff``.

    Args:
        mol (pyscf.gto.Mole):
            Molecular geometry and basis set.
        mf (pyscf.scf.hf.SCF):
            Converged mean-field object supplying ``mo_coeff`` and ``mo_occ``.
        n_occ (int, optional):
            Number of occupied orbitals. Inferred from *mf* if not given.
        minao (str):
            Reference minimal basis (default ``"minao"``).

    Returns:
        iaos (np.ndarray, shape (nao, n_iao)):
            Löwdin-orthogonalized IAO coefficients, with ``n_iao`` the size of the
            *minao* basis (``n_iao >= n_occ``).
    """
    from pyscf.lo import iao as _iao  # noqa: PLC0415
    from pyscf.lo import orth as _orth  # noqa: PLC0415

    if n_occ is None:
        n_occ = int(np.count_nonzero(mf.mo_occ > 0))
    iaos = _iao.iao(mol, mf.mo_coeff[:, :n_occ], minao=minao)
    s = mol.intor_symmetric("int1e_ovlp")
    return np.asarray(_orth.vec_lowdin(iaos, s))

    def localize_fnos(
    mf: Any,
    correlated_method: Literal["mp2", "cisd", "ccsd"],
    active: bool | list[int] = True,  # noqa: FBT001, FBT002
    *,
    frozen: list[int] | None = None,
) -> np.ndarray:
    """Frozen natural orbital (FNO) transformation.

    Builds a correlated wavefunction, partitions orbitals into a frozen core and
    an active set, and returns the FNO coefficient matrix.

    Args:
        mf (pyscf.scf.hf.SCF):
            Converged mean-field object.
        correlated_method ({'mp2', 'cisd', 'ccsd'}):
            Correlated method used to build the 1-RDM for the NO transformation.
        active (True or list of int):
            Controls which occupied orbitals enter the correlated calculation.
            ``True`` uses all occupied orbitals (no freezing). A list of integers
            specifies the indices of the active occupied orbitals; all others
            are frozen. Ignored when *frozen* is provided. Indices are given in
            the configured display base (see :func:`aegiss.utils.orb_display.orbital_base`):
            0-based by default, or 1-based when ``AEGISS_ORB_BASE=1``.
        frozen (list of int, optional):
            Indices of occupied orbitals to freeze, in the configured display
            base (0-based by default; 1-based when ``AEGISS_ORB_BASE=1``). Takes
            precedence over *active* when provided.

    Returns:
        mo_coeff (np.ndarray, shape (nao, nmo)):
            FNO coefficients in the AO basis.
    """
    from aegiss.localization.fnos import compute_fnos  # noqa: PLC0415

    logger.info("Computing FNOs using %s", correlated_method.upper())

    orbital_list: list[int]
    if frozen is not None:
        # `frozen` is user input in the display base; convert to 0-based first.
        # `frozen` names the orbitals to exclude; active for get_orbital_spaces
        # is the complement.
        frozen_set = set(from_display_all(frozen))
        orbital_list = [
            int(i) for i in np.nonzero(mf.mo_occ > 0)[0] if i not in frozen_set
        ]
    elif isinstance(active, list):
        # `active` is user input in the display base; convert to 0-based.
        orbital_list = from_display_all(active)
    elif active is True:
        orbital_list = []
    else:
        raise ValueError(
            "active=False is not supported; pass True to use all occupied orbitals or a list of active orbital indices."
        )
    frozen_orbs, active_orbs, _, frozen_virt = get_orbital_spaces(
        mf,
        orbital_list=orbital_list,
        unique_virt_space=True,
    )
    wfn_obj = build_correlated_wfn(mf, correlated_method, frozen=frozen_orbs)
    logger.info("Frozen orbitals: %s", to_display_all(frozen_orbs))
    logger.info("Active orbitals: %s", to_display_all(active_orbs))
    logger.info("Frozen virtual orbitals: %s", to_display_all(frozen_virt))

    return compute_fnos(
        wfn_obj,
        frozen=frozen_orbs.tolist(),
        active=active_orbs.tolist(),
    )

    def compute_nos(wfn_obj: Any, *, split_nos: bool = False) -> np.ndarray:
    """Compute natural orbitals from a correlated wavefunction object.

    Extracts the 1-RDM from *wfn_obj* (CISD, CCSD, CASCI/CASSCF, …),
    diagonalises it, and returns the NOs sorted by descending occupation.

    Args:
        wfn_obj (correlated PySCF solver object):
            Must expose a ``make_rdm1`` method returning the 1-RDM in the MO basis.
        split_nos (bool, optional):
            Obtain the Natural Orbital basis by diagonalizing the 1-RDM in the
            occupied and virtual subspaces separately. Useful for certain post-HF
            methods where the occupied and virtual spaces are treated differently.

    Returns:
        nos (np.ndarray, shape (nao, nmo)):
            Natural orbital coefficients in the AO basis.
    """
    rdm1_mo = wfn_obj.make_rdm1()
    logger.debug("1-RDM in MO basis:\n%s", rdm1_mo.shape)
    if split_nos:
        logger.info(
            "Building Split NOs by diagonalizing occupied and virtual blocks separately."
        )
        nocc = wfn_obj.nocc
        rdm1_occ = rdm1_mo[:nocc, :nocc]
        rdm1_virt = rdm1_mo[nocc:, nocc:]

        occ_occ, u_occ = np.linalg.eigh(rdm1_occ)
        occ_virt, u_virt = np.linalg.eigh(rdm1_virt)

        # Sort occupied and virtual NOs separately
        idx_occ = np.argsort(occ_occ)[::-1]
        idx_virt = np.argsort(occ_virt)[::-1]

        u_occ = u_occ[:, idx_occ]
        u_virt = u_virt[:, idx_virt]

        # Combine occupied and virtual NOs back into a single MO coefficient matrix
        nos_occ = wfn_obj.mo_coeff[:, :nocc] @ u_occ
        nos_virt = wfn_obj.mo_coeff[:, nocc:] @ u_virt
        return np.hstack((nos_occ, nos_virt))
    logger.info("Building standard NOs by diagonalizing the full 1-RDM.")
    occ_mo, u = np.linalg.eigh(rdm1_mo)
    # sort descending
    idx = np.argsort(occ_mo)[::-1]
    u = u[:, idx]
    # pyscf post-HF objects expose their SCF only as `_scf`
    scf = getattr(wfn_obj, "_scf", wfn_obj)
    mo_coeff = scf.mo_coeff
    return np.asarray(mo_coeff @ u)

    def compute_fnos(
    wfn_obj: Any,
    frozen: list[int],
    active: list[int],
) -> np.ndarray:
    """Construct frozen natural orbitals by rotating the virtual space into NOs.

    Only the virtual block of the 1-RDM is diagonalised; the virtual NOs are
    sorted by decreasing occupation. The occupied orbitals are returned
    unchanged and no orbital is discarded.

    Args:
        wfn_obj (correlated PySCF solver object):
            Must expose a ``make_rdm1`` method returning the 1-RDM in the MO basis.
        frozen (list of int):
            Indices of orbitals to be frozen (0-based indexing).
        active (list of int):
            Indices of orbitals to be retained in the active space (0-based indexing).

    Returns:
        mo_coeff (np.ndarray, shape (nao, nmo)):
            Unchanged occupied (frozen and active) orbitals followed by the
            virtual FNOs, in the AO basis.
    """
    rdm1 = wfn_obj.make_rdm1()

    nocc = len(frozen) + len(active)
    logger.debug("1-RDM in MO basis:\n%s", rdm1.shape)
    logger.info(
        "Diagonalizing virtual block of the 1-RDM to construct FNOs for virtual orbitals."
    )
    dm_vv = rdm1[nocc:, nocc:]
    occ_vv, u_vv = np.linalg.eigh(dm_vv)
    idx_vv = np.argsort(occ_vv)[::-1]
    occ_vv = occ_vv[idx_vv]
    u_vv = u_vv[:, idx_vv]

    mo_coeff_vv = wfn_obj.mo_coeff[:, nocc:] @ u_vv
    mo_coeff_oo = wfn_obj.mo_coeff[:, :nocc]
    return np.hstack((mo_coeff_oo, mo_coeff_vv))
