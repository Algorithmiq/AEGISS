    Wavefunction

    Correlated wavefunction methods. run_dmrg_subprocess is the primary method used in the AEGISS pipeline; the others (run_mp2, run_cisd, run_ccsd) are thin wrappers around PySCF that produce correlated density matrices for natural orbital localization.

    def run_dmrg_subprocess(
    mf: Any,
    ncas: int,
    nelecas: int | tuple[int, int],
    max_bond_dim: int = 500,
    symm_type: str = "SZ",
    verbosity: int = 0,
    *,
    unrestricted: bool | None = None,
    get_qmi: bool = True,
    spin_resolved: bool = False,
    reorder: str = "noreorder",
) -> tuple[float, np.ndarray | None]:
    """Run a DMRG calculation via the ``block2main`` subprocess.

    Args:
        mf (pyscf.scf.hf.SCF): Converged mean-field object.
        ncas (int): Number of active orbitals.
        nelecas (int or tuple[int, int]): Number of active electrons (total
            count or ``(nalpha, nbeta)`` pair).
        max_bond_dim (int): Target bond dimension for the DMRG schedule (must
            be a positive multiple of 4).
        symm_type ({'SU2', 'SZ'}): Spin symmetry: ``'SZ'`` adds the
            ``nonspinadapted`` keyword so block2main runs without spin
            adaptation; ``'SU2'`` omits it.
        verbosity (int): If non-zero, block2main stdout/stderr are forwarded to
            the terminal.
        unrestricted (bool or None): Whether to emit an unrestricted (UHF,
            ``IUHF=1``) FCIDUMP with separate aa/bb/ab and alpha/beta blocks. If
            ``None`` (default), autodetected from ``isinstance(mf, scf.uhf.UHF)``;
            pass ``True``/``False`` to override. UHF requires ``symm_type='SZ'``.
        get_qmi (bool): If ``True``, compute and return single-orbital
            entropies.
        spin_resolved (bool): If ``True``, return separate alpha/beta
            spin-orbital entropies as a ``(2, ncas)`` array (row 0 = alpha,
            row 1 = beta) instead of the spin-summed ``(ncas,)`` vector. Only
            meaningful with *get_qmi*. The spin-summed vector feeds
            :func:`~aegiss.active_space.entropy_screening.select_by_entropy`;
            the ``(2, ncas)`` array feeds
            :func:`~aegiss.active_space.entropy_screening.select_by_spin_entropy`.
            :func:`~aegiss.active_space.aegiss.select_aegiss` sets this from
            :meth:`DMRGConfig.spin_resolved`.
        reorder ({'noreorder', 'fiedler'}): Orbital ordering for block2main.
            ``'noreorder'`` (default) keeps the FCIDUMP order and enables
            ``full_fci_space``; ``'fiedler'`` reorders by the Fiedler vector and
            disables ``full_fci_space`` (that combination can converge to a
            higher local minimum). Output ``1pdm``/``1npc`` are mapped back to
            the input order either way, so per-orbital entropy indexing is
            unaffected.

    Returns:
        energy (float): DMRG ground-state energy.
        soe1 (np.ndarray or None): Single-orbital entropies (``None`` if
            *get_qmi* is ``False``). Shape ``(ncas,)`` by default, or
            ``(2, ncas)`` if *spin_resolved* is ``True``. Two-orbital entropies
            and mutual information are not available from the ``1pdm`` +
            ``1npc`` output; pass them from an external source to any downstream
            plotting functions if needed.

    Raises:
        ValueError: If *symm_type* or *reorder* is not a supported value, if
            *unrestricted* is used with ``symm_type='SU2'``, or if
            *max_bond_dim* is not a positive multiple of 4.
        RuntimeError: If the ``block2main`` executable cannot be found.
        subprocess.CalledProcessError: If ``block2main`` exits with a non-zero
            status.
    """
    if symm_type not in _VALID_SYMM_TYPES:
        raise ValueError(
            f"Unsupported symmetry type: {symm_type!r}. Expected one of {list(_VALID_SYMM_TYPES)}."
        )
    if reorder not in _VALID_REORDER:
        raise ValueError(
            f"Unsupported reorder: {reorder!r}. Expected one of {list(_VALID_REORDER)}."
        )

    if unrestricted is None:
        unrestricted = isinstance(mf, scf.uhf.UHF)
    if unrestricted and symm_type != "SZ":
        raise ValueError(
            f"Unrestricted (UHF) integrals require symm_type='SZ', got {symm_type!r}. "
            "block2 cannot read an IUHF=1 FCIDUMP in a spin-adapted (SU2) run."
        )

    scratch = get_scratch_dir()
    n_threads = omp_num_threads()
    stack_mem_gb = mem() / (1024**3)

    bond_dims, noises, thrds = _default_dmrg_schedule(max_bond_dim)
    nalpha, nbeta = resolve_nelecas(nelecas)
    n_sweeps = len(bond_dims)
    twosite_to_onesite = max(n_sweeps - 2, 0)

    logger.info(
        "----------------DMRG Calculation (symmetry: %s)----------------\n", symm_type
    )
    logger.info(
        "Using pyscf.%s to obtain integrals for DMRG...",
        "UCASCI" if unrestricted else "CASCI",
    )
    logger.info("CAS (ncas=%d, nelecas=%s)", ncas, nelecas)
    logger.info(
        "System parameters: n_sites=%d, n_elec=%d, spin=%d, symm_type=%s",
        ncas,
        nalpha + nbeta,
        nalpha - nbeta,
        symm_type,
    )
    logger.info("Scratch directory: %s", scratch)
    logger.info("Number of threads: %d", n_threads)
    logger.info(
        "DMRG settings: n_sweeps = %d, max_bond_dim = %d, twosite_to_onesite = %d",
        n_sweeps,
        max_bond_dim,
        twosite_to_onesite,
    )
    logger.info("DMRG Noise: init = %s, final = %s", noises[0], noises[-1])
    logger.info("DMRG Threshold: init = %s, final = %s\n", thrds[0], thrds[-1])

    work_dir = Path(scratch) / f"dmrg_subprocess_{time.time_ns()}"
    work_dir.mkdir(parents=True, exist_ok=False)
    fcidump_path = work_dir / "FCIDUMP"
    if unrestricted:
        _write_uhf_fcidump(fcidump_path, mf, ncas, (nalpha, nbeta))
    else:
        _write_fcidump(fcidump_path, mf, ncas, nelecas)

    input_path = work_dir / "dmrg.conf"
    _write_input(
        input_path,
        fcidump_path,
        nalpha + nbeta,
        nalpha - nbeta,
        ncas,
        work_dir,
        bond_dims,
        noises,
        thrds,
        n_threads,
        stack_mem_gb,
        symm_type,
        reorder,
    )

    _venv_bin = Path(sys.executable).parent
    block2main_path = shutil.which("block2main") or (
        str(_venv_bin / "block2main") if (_venv_bin / "block2main").exists() else None
    )
    if block2main_path is None:
        raise RuntimeError(
            "block2main not found in PATH or venv bin directory. "
            "Activate the project venv or add block2main to PATH."
        )

    logger.info("Running block2main: %s %s", block2main_path, input_path)
    t = time.time()
    subprocess.run(  # noqa: S603
        [block2main_path, str(input_path)],
        capture_output=(verbosity == 0),
        check=True,
    )
    logger.info("DMRG completed in %.3f [s]", time.time() - t)

    energy, pdm1, npc1 = _read_outputs(work_dir, ncas)
    logger.info("DMRG Energy (%s)\t : %.8f", symm_type, energy)

    if not get_qmi:
        return energy, None

    soe1 = _compute_spin_soe(pdm1) if spin_resolved else _compute_soe(pdm1, npc1)
    return energy, soe1

    def run_mp2(mf: Any, **kwargs: Any) -> Any:
    """Run MP2 on top of a mean-field reference.

    Args:
        mf (pyscf.scf.hf.SCF): Converged mean-field object.
        **kwargs: Additional attributes forwarded to the PySCF MP2 object.

    Returns:
        mymp2 (pyscf.mp.MP2): Converged MP2 object.
    """
    logger.info("Running MP2 with options: %s", kwargs)
    mymp2 = mp.MP2(mf)
    for key, val in kwargs.items():
        setattr(mymp2, key, val)
    mymp2.kernel()
    return mymp2
